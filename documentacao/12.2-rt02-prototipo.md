<script>
  // Caminho relativo partindo de galeria/index.html no repositório.
  // Não é preciso informar nome de branch ou endereço Raw.
  // Estrutura: /galeria/index.html e /documentação/Imagens/imagem1.png ... imagem10.png
  const IMAGE_BASE = '../documentação/Imagens/';

  const titles = [
    'Boas-vindas',
    'Cadastro de usuário',
    'Entrevista guiada',
    'Currículo gerado por IA',
    'Preferências profissionais',
    'Vagas recomendadas',
    'Detalhes da vaga',
    'Confirmação de candidatura',
    'Acompanhamento de candidaturas',
    'Conversa com a empresa'
  ];

  const screen = document.getElementById('screen');
  const count = document.getElementById('count');
  const title = document.getElementById('title');
  const prev = document.getElementById('previous');
  const next = document.getElementById('next');
  const dots = document.getElementById('dots');
  const error = document.getElementById('error');
  let current = 0;

  for (let i = 0; i < titles.length; i++) {
    const button = document.createElement('button');
    button.type = 'button';
    button.className = 'dot';
    button.setAttribute('aria-label', `Ir para a tela ${i + 1}: ${titles[i]}`);
    button.addEventListener('click', () => show(i));
    dots.appendChild(button);
  }

  function show(i) {
    current = Math.min(titles.length - 1, Math.max(0, i));
    error.textContent = '';
    screen.src = `${IMAGE_BASE}imagem${current + 1}.png`;
    screen.alt = `Tela ${String(current + 1).padStart(2, '0')} — ${titles[current]}`;
    title.textContent = titles[current];
    count.textContent = `${current + 1} de ${titles.length}`;
    prev.disabled = current === 0;
    next.disabled = current === titles.length - 1;
    [...dots.children].forEach((button, index) => {
      button.setAttribute('aria-current', String(index === current));
    });
    history.replaceState(null, '', `#tela-${String(current + 1).padStart(2, '0')}`);
  }

  screen.addEventListener('error', () => {
    error.textContent = 'Não foi possível carregar esta imagem. Confira o nome, a pasta, a branch e a visibilidade do repositório.';
  });
  prev.addEventListener('click', () => show(current - 1));
  next.addEventListener('click', () => show(current + 1));
  document.addEventListener('keydown', (event) => {
    if (event.key === 'ArrowLeft') show(current - 1);
    if (event.key === 'ArrowRight') show(current + 1);
  });

  const initial = Number((location.hash.match(/^#tela-(\d+)$/) || [])[1]);
  show(initial >= 1 && initial <= titles.length ? initial - 1 : 0);
</script>
